def write_json(path, value):
    Path(path).write_text(
        json.dumps(value, indent=2, default=str) + "\n"
    )


def read_exact(sock, length):
    parts = []
    remaining = length
    while remaining:
        data = sock.recv(remaining)
        if not data:
            raise EOFError("Connection closed before frame completed")
        parts.append(data)
        remaining -= len(data)
    return b"".join(parts)


def send_json(sock, value):
    payload = json.dumps(value).encode("utf-8")
    if len(payload) > 65536:
        raise ValueError("JSON frame too large")
    sock.sendall(struct.pack("!I", len(payload)) + payload)


def receive_json(sock):
    length = struct.unpack("!I", read_exact(sock, 4))[0]
    if length < 2 or length > 65536:
        raise ValueError(f"Invalid JSON frame length: {length}")
    value = json.loads(read_exact(sock, length))
    if not isinstance(value, dict):
        raise ValueError("Expected JSON object")
    return value

def synchronize(torch, device):

        torch.cuda.synchronize(device)

def snn_components(device):
    saved_argv = sys.argv[:]

        import snn.snn_snasnet.eval_snasnet_cifar10 as api
        model, config, _ = api.load_snn_checkpoint(device=device)
    finally:
        sys.argv = saved_argv

    from spikingjelly.clock_driven import functional

    _, transform = api.data_transforms(config)

    def reset():
        functional.reset_net(model)
        model.neuron_init()

def infer(torch, device, model, transform, image, reset=None):
    if reset is not None:
        reset()

    batch = transform(image).unsqueeze(0).to(device)
    scores = model(batch)
    confidence, prediction = torch.softmax(
        scores.float(), dim=1
    ).max(dim=1)

    prediction = int(prediction.item())
    confidence = float(confidence.item())
    synchronize(torch, device)

    if tuple(scores.shape) != (1, 10):
        raise ValueError(f"Unexpected scores: {tuple(scores.shape)}")
    if not torch.isfinite(scores).all().item():
        raise ValueError("Non-finite model output")

    return prediction, confidence



def verify_worker_network(args, expected_ip):
    namespace = os.readlink("/proc/self/ns/net")
    expected_namespace = os.environ.get("NSSON_EXPECTED_NETNS")

    result = subprocess.run(
        ["ip", "-j", "-4", "address", "show"],
        text=True,
        capture_output=True,
        timeout=10,
    )

    if result.returncode != 0:
        raise RuntimeError(
            f"Cannot inspect worker addresses: {result.stderr}"
        )

    interfaces = json.loads(result.stdout)
    addresses = sorted({
        address["local"]
        for interface in interfaces
        for address in interface.get("addr_info", [])
        if address.get("family") == "inet"
        and "local" in address
    })

    audit = {
        "role": args.role,
        "pid": os.getpid(),
        "uid": os.getuid(),
        "gid": os.getgid(),
        "actual_network_namespace": namespace,
        "expected_network_namespace": expected_namespace,
        "expected_ip": expected_ip,
        "observed_ipv4_addresses": addresses,
        "namespace_matches": namespace == expected_namespace,
        "expected_ip_present": expected_ip in addresses,
    }

    audit_path = (
        Path(args.out) / f"{args.role}_namespace.json"
    )
    write_json(audit_path, audit)
    print(json.dumps(audit, indent=2), flush=True)

    if os.getuid() == 0:
        raise RuntimeError(
            "Worker unexpectedly retained root credentials"
        )

    if not expected_namespace:
        raise RuntimeError("Missing expected namespace identity")

    if namespace != expected_namespace:
        raise RuntimeError(
            "Worker is in the wrong network namespace. "
            "Stopped before model loading."
        )

    if expected_ip not in addresses:
        raise RuntimeError(
            f"Worker does not own {expected_ip}. "
            "Stopped before model loading."
        )

    # Test address availability without occupying the inference port.
    with socket.socket(
        socket.AF_INET, socket.SOCK_STREAM
    ) as probe:
        probe.bind((expected_ip, 0))
        audit["temporary_bind_test_passed"] = True
        audit["temporary_bound_address"] = list(
            probe.getsockname()
        )

    write_json(audit_path, audit)
    print(
        f"{args.role.upper()} namespace and address checks passed",
        flush=True,
    )

def run_server(args):
    verify_worker_network(args, SERVER_IP)
    _, torch, device = tensor_environment()
    from PIL import Image

    print("STAGE: before CNN loading", flush=True)
    model, transform = cnn_components(device)
    print("STAGE: CNN loading completed", flush=True)

    print("STAGE: before server warm-up", flush=True)
    with torch.no_grad():
        warm_image = Image.new("RGB", (32, 32))
        for _ in range(10):
            infer(torch, device, model, transform, warm_image)

    print("STAGE: server warm-up completed", flush=True)
    out = Path(args.out)
    fields = [
        "run_id", "session_id", "request_id", "sample_id",
        "client_ip", "client_port", "server_ip", "server_port",
        "image_sha256", "prediction",
        "confidence_uncalibrated", "server_inference_sec",
    ]

    seen = set()

    with (out / "server_requests.csv").open("w", newline="") as log:
        writer = csv.DictWriter(log, fieldnames=fields)
        writer.writeheader()
        log.flush()

        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as listener:
            listener.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
            print(
                f"STAGE: before bind {SERVER_IP}:{SERVER_PORT}; "
                f"netns={os.readlink(chr(47) + 'proc/self/ns/net')}",
                flush=True,
            )
            listener.bind((SERVER_IP, SERVER_PORT))
            print("STAGE: server bind completed", flush=True)
            listener.listen(8)

            write_json(out / "server_ready.json", {
                "server_ip": SERVER_IP,
                "server_port": SERVER_PORT,
                "device": str(device),
                "scope": "Workstation CNN inside Mininet server namespace",
            })

            print("SERVER READY", flush=True)

            while True:
                connection, peer = listener.accept()

                with connection:
                    connection.settimeout(60)
                    while True:
                        try:
                            header = receive_json(connection)
                        except EOFError:
                            break

                        for key in (
                            "run_id", "session_id",
                            "request_id", "sample_id",
                            "payload_bytes", "image_sha256",
                        ):
                            if key not in header:
                                raise ValueError(f"Missing field: {key}")

                        if header["run_id"] != out.name:
                            raise ValueError("Wrong run identity")
                        if header["payload_bytes"] != 3072:
                            raise ValueError("Expected one raw RGB 32x32 image")

                        payload = read_exact(connection, 3072)
                        digest = hashlib.sha256(payload).hexdigest()
                        if digest != header["image_sha256"]:
                            raise ValueError("Payload checksum mismatch")

                        identity = (
                            header["run_id"],
                            header["session_id"],
                            header["request_id"],
                        )
                        if identity in seen:
                            raise ValueError("Duplicate request identity")
                        seen.add(identity)

                        image = Image.frombytes("RGB", (32, 32), payload)

                        synchronize(torch, device)
                        start = time.perf_counter_ns()
                        with torch.no_grad():
                            prediction, confidence = infer(
                                torch, device, model, transform, image
                            )
                        end = time.perf_counter_ns()

                        response = {
                            "run_id": header["run_id"],
                            "session_id": header["session_id"],
                            "request_id": header["request_id"],
                            "sample_id": int(header["sample_id"]),
                            "image_sha256": digest,
                            "prediction": prediction,
                            "confidence_uncalibrated": confidence,
                            "server_inference_sec": (end - start) / 1e9,
                            "client_ip": peer[0],
                            "client_port": peer[1],
                            "server_ip": SERVER_IP,
                            "server_port": SERVER_PORT,
                        }

                        writer.writerow(response)
                        log.flush()
                        send_json(connection, response)


def run_client(args):
    verify_worker_network(args, CLIENT_IP)
    np, torch, device = tensor_environment()
    import torchvision

    out = Path(args.out)
    cnn, cnn_transform = cnn_components(device)
    snn, snn_transform, reset_snn = snn_components(device)

    dataset = torchvision.datasets.CIFAR10(
        root=str(ROOT / "data/cifar10"),
        train=False,
        download=False,
        transform=None,
    )

    ids = np.random.default_rng(20261102).permutation(
        len(dataset)
    )[:REQUESTS_PER_POLICY].tolist()

    samples = {}
    for sample_id in ids:
        image, label = dataset[sample_id]
        payload = image.tobytes()
        if len(payload) != 3072:
            raise ValueError("Unexpected raw image size")
        samples[sample_id] = (
            image, int(label), payload,
            hashlib.sha256(payload).hexdigest(),
        )

    # Warm-up both local model paths outside recorded requests.
    with torch.no_grad():
        for sample_id in ids[:10]:
            image = samples[sample_id][0]
            infer(torch, device, cnn, cnn_transform, image)
            infer(
                torch, device, snn, snn_transform, image,
                reset=reset_snn,
            )

    fields = [
        "run_id", "policy", "sample_id", "label",
        "decision", "threshold", "snn_confidence_uncalibrated",
        "final_prediction", "correct",
        "local_phase_sec", "remote_rpc_sec",
        "request_sec", "residual_overhead_sec",
        "session_id", "request_id",
        "client_ip", "client_port", "server_ip", "server_port",
        "image_sha256", "server_inference_sec",
    ]

    write_json(out / "client_configuration.json", {
        "policies": POLICIES,
        "requests_per_policy": REQUESTS_PER_POLICY,
        "sample_ids": ids,
        "threshold": THETA,
        "device": str(device),
        "scope": (
            "Two network namespaces on the same workstation. "
            "Shared physical CPU/GPU resources. No Jetson measurements."
        ),
        "timing_scope": (
            "Original image retrieval and raw-payload preparation excluded. "
            "Local phase includes preprocessing, reset where applicable, "
            "inference, and prediction extraction. Remote RPC includes "
            "connection establishment, transport, server execution, and response."
        ),
        "limitations": [
            "Smoke test, not a final repeated benchmark.",
            "Static OpenFlow forwarding; no Ryu association or QoS queues.",
            "Fixed threshold; no adaptive feedback.",
            "No energy model applied.",
        ],
    })

    with (out / "client_requests.csv").open("w", newline="") as log:
        writer = csv.DictWriter(log, fieldnames=fields)
        writer.writeheader()

        for policy in POLICIES:
            session_id = str(uuid.uuid4())
            connection = None

            print(f"CLIENT POLICY: {policy}", flush=True)

            try:
                with torch.no_grad():
                    for sequence, sample_id in enumerate(ids):
                        image, label, payload, digest = samples[sample_id]

                        local_sec = 0.0
                        rpc_sec = 0.0
                        snn_confidence = ""
                        response = None

                        synchronize(torch, device)
                        start = time.perf_counter_ns()

                        if policy == "local_cnn":
                            phase_start = time.perf_counter_ns()
                            prediction, _ = infer(
                                torch, device,
                                cnn, cnn_transform, image,
                            )
                            local_sec = (
                                time.perf_counter_ns() - phase_start
                            ) / 1e9
                            offload = False

                        elif policy in ("local_snn", "adaptive_fixed"):
                            phase_start = time.perf_counter_ns()
                            prediction, snn_confidence = infer(
                                torch, device,
                                snn, snn_transform, image,
                                reset=reset_snn,
                            )
                            local_sec = (
                                time.perf_counter_ns() - phase_start
                            ) / 1e9
                            offload = (
                                policy == "adaptive_fixed"
                                and snn_confidence < THETA
                            )

                        else:
                            # Explicit offload-only: not theta=1.
                            offload = True

                        request_id = f"{policy}:{sequence}:{sample_id}"

                        if offload:
                            rpc_start = time.perf_counter_ns()

                            if connection is None:
                                connection = socket.create_connection(
                                    (SERVER_IP, SERVER_PORT),
                                    timeout=60,
                                )
                                connection.settimeout(60)

                            header = {
                                "run_id": out.name,
                                "session_id": session_id,
                                "request_id": request_id,
                                "sample_id": sample_id,
                                "payload_bytes": len(payload),
                                "image_sha256": digest,
                            }

                            send_json(connection, header)
                            connection.sendall(payload)
                            response = receive_json(connection)

                            for key in (
                                "run_id", "session_id", "request_id",
                                "sample_id", "image_sha256",
                            ):
                                if response.get(key) != header[key]:
                                    raise RuntimeError(
                                        f"Response association mismatch: {key}"
                                    )

                            local_address = connection.getsockname()
                            if (
                                response["client_ip"] != local_address[0]
                                or response["client_port"] != local_address[1]
                            ):
                                raise RuntimeError(
                                    "Client/server transport tuple mismatch"
                                )

                            prediction = int(response["prediction"])
                            rpc_sec = (
                                time.perf_counter_ns() - rpc_start
                            ) / 1e9

                        total_sec = (
                            time.perf_counter_ns() - start
                        ) / 1e9

                        row = {
                            "run_id": out.name,
                            "policy": policy,
                            "sample_id": sample_id,
                            "label": label,
                            "decision": "offload" if offload else "local",
                            "threshold": (
                                THETA if policy == "adaptive_fixed" else ""
                            ),
                            "snn_confidence_uncalibrated": snn_confidence,
                            "final_prediction": prediction,
                            "correct": int(prediction == label),
                            "local_phase_sec": local_sec,
                            "remote_rpc_sec": rpc_sec,
                            "request_sec": total_sec,
                            "residual_overhead_sec": (
                                total_sec - local_sec - rpc_sec
                            ),
                            "session_id": (
                                session_id if offload else ""
                            ),
                            "request_id": (
                                request_id if offload else ""
                            ),
                            "client_ip": (
                                response["client_ip"] if offload else ""
                            ),
                            "client_port": (
                                response["client_port"] if offload else ""
                            ),
                            "server_ip": (
                                response["server_ip"] if offload else ""
                            ),
                            "server_port": (
                                response["server_port"] if offload else ""
                            ),
                            "image_sha256": digest,
                            "server_inference_sec": (
                                response["server_inference_sec"]
                                if offload else ""
                            ),
                        }

                        writer.writerow(row)
                        log.flush()

            finally:
                if connection is not None:
                    connection.close()

    print("CLIENT COMPLETED", flush=True)


def run_topology(args):
    if os.geteuid() != 0:
        raise RuntimeError("Topology creation requires sudo")

    from mininet.net import Mininet
    from mininet.node import OVSSwitch
    from mininet.link import TCLink
    from mininet.log import setLogLevel

    setLogLevel("info")

    out = Path(args.out).resolve()
    owner = pwd.getpwnam(args.user)
    setpriv = shutil.which("setpriv")

    if setpriv is None:
        raise RuntimeError("setpriv is unavailable")

    suffix = uuid.uuid4().hex[:6]
    bridge = "nr" + suffix + "s"
    client_name = "nr" + suffix + "c"
    server_name = "nr" + suffix + "r"

    def command(argv):
        result = subprocess.run(
            argv, text=True, capture_output=True, timeout=20
        )
        if result.returncode != 0:
            raise RuntimeError(
                f"Command failed: {argv}\n"
                f"{result.stdout}\n{result.stderr}"
            )
        return result.stdout

    if subprocess.run(
        ["ovs-vsctl", "br-exists", bridge],
        capture_output=True,
    ).returncode == 0:
        raise RuntimeError(f"Refusing to reuse bridge: {bridge}")

    def user_command(role):
        if role not in ("server", "client"):
            raise ValueError(f"Unexpected worker role: {role}")

        endpoint = server if role == "server" else client
        expected_namespace = os.readlink(
            f"/proc/{endpoint.pid}/ns/net"
        )

        return [
            setpriv,
            "--reuid", str(owner.pw_uid),
            "--regid", str(owner.pw_gid),
            "--init-groups",
            "env",
            f"HOME={owner.pw_dir}",
            f"USER={args.user}",
            f"LOGNAME={args.user}",
            f"PYTHONPATH={ROOT}",
            f"NSSON_EXPECTED_NETNS={expected_namespace}",
            "PYTHONUNBUFFERED=1",
            "PYTHONFAULTHANDLER=1",
            "MPLBACKEND=Agg",
            str(PYTHON), "-u", str(SCRIPT),
            "--role", role,
            "--out", str(out),
        ]

    def save_owned(name, text):
        path = out / name
        path.write_text(text)
        os.chown(path, owner.pw_uid, owner.pw_gid)

    network = None
    processes = []
    log_handles = []

    try:
        network = Mininet(
            controller=None,
            switch=OVSSwitch,
            link=TCLink,
            build=False,
            autoSetMacs=True,
        )

        client = network.addHost(
            client_name, ip=CLIENT_IP + "/24"
        )
        server = network.addHost(
            server_name, ip=SERVER_IP + "/24"
        )
        switch = network.addSwitch(
            bridge,
            protocols="OpenFlow13",
            failMode="secure",
        )

        network.addLink(
            client, switch,
            port1=0, port2=1,
            bw=20, delay="2ms",
        )
        network.addLink(
            server, switch,
            port1=0, port2=2,
            bw=20, delay="2ms",
        )

        network.build()
        network.start()

        controllers = command([
            "ovs-vsctl", "get-controller", bridge
        ]).strip()
        if controllers:
            raise RuntimeError(
                f"Unexpected controller on isolated bridge: {controllers}"
            )

        # Explicit, static bidirectional baseline.
        # These are NOT QoS queue-selection rules.
        for rule in (
            "priority=10,in_port=1,actions=output:2",
            "priority=10,in_port=2,actions=output:1",
        ):
            command([
                "ovs-ofctl", "-O", "OpenFlow13",
                "add-flow", bridge, rule,
            ])

        save_owned("flows_before.txt", command([
            "ovs-ofctl", "-O", "OpenFlow13",
            "dump-flows", bridge,
        ]))

        ping = client.cmd(
            f"ping -c 2 -W 3 {SERVER_IP}"
        )
        print(ping, flush=True)
        save_owned("connectivity.txt", ping)

        if " 0% packet loss" not in ping:
            raise RuntimeError("Isolated network connectivity failed")

        save_owned(
            "network_configuration.json",
            json.dumps({
                "bridge": bridge,
                "controller": None,
                "forwarding": "Static OpenFlow 1.3 baseline",
                "client": CLIENT_IP,
                "server": SERVER_IP,
                "each_link_configured_mbps": 20,
                "each_link_configured_delay": "2ms",
                "warning": (
                    "Configured emulation settings are not measured "
                    "latency or bandwidth guarantees."
                ),
            }, indent=2) + "\n",
        )

        server_log = (out / "server_console.log").open("w")
        log_handles.append(server_log)
        os.chown(
            out / "server_console.log",
            owner.pw_uid, owner.pw_gid,
        )

        server_process = server.popen(
            user_command("server"),
            stdout=server_log,
            stderr=subprocess.STDOUT,
            start_new_session=True,
        )
        processes.append(server_process)

        deadline = time.monotonic() + 180
        while not (out / "server_ready.json").is_file():
            server_returncode = server_process.poll()
            if server_returncode is not None:
                if server_returncode < 0:
                    try:
                        explanation = signal.Signals(
                            -server_returncode
                        ).name
                    except ValueError:
                        explanation = (
                            f"signal {-server_returncode}"
                        )
                else:
                    explanation = (
                        f"ordinary exit status {server_returncode}"
                    )

                save_owned(
                    "server_exit.json",
                    json.dumps({
                        "pid": server_process.pid,
                        "returncode": server_returncode,
                        "interpretation": explanation,
                    }, indent=2) + "\n",
                )

                raise RuntimeError(
                    f"Server exited: returncode={server_returncode}; "
                    f"{explanation}. Inspect server_console.log."
                )
            if time.monotonic() > deadline:
                raise TimeoutError(
                    "Server startup timeout. Inspect server_console.log"
                )
            time.sleep(0.5)

        client_log = (out / "client_console.log").open("w")
        log_handles.append(client_log)
        os.chown(
            out / "client_console.log",
            owner.pw_uid, owner.pw_gid,
        )

        client_process = client.popen(
            user_command("client"),
            stdout=client_log,
            stderr=subprocess.STDOUT,
            start_new_session=True,
        )
        processes.append(client_process)

        try:
            returncode = client_process.wait(timeout=900)
        except subprocess.TimeoutExpired:
            raise TimeoutError(
                "Client timeout. Inspect client_console.log"
            )

        if returncode != 0:
            save_owned(
                "client_exit.json",
                json.dumps({
                    "pid": client_process.pid,
                    "returncode": returncode,
                }, indent=2) + "\n",
            )
            raise RuntimeError(
                f"Client failed: returncode={returncode}. "
                "Inspect client_console.log."
            )

        save_owned("flows_after.txt", command([
            "ovs-ofctl", "-O", "OpenFlow13",
            "dump-flows", bridge,
        ]))
        save_owned("ports_after.txt", command([
            "ovs-ofctl", "-O", "OpenFlow13",
            "dump-ports", bridge,
        ]))

        with (out / "client_requests.csv").open() as f:
            rows = list(csv.DictReader(f))

        with (out / "server_requests.csv").open() as f:
            server_rows = list(csv.DictReader(f))

        if len(rows) != REQUESTS_PER_POLICY * len(POLICIES):
            raise RuntimeError("Incomplete client request log")

        expected = [
            row for row in rows
            if row["decision"] == "offload"
        ]

        def identity(row):
            return (
                row["run_id"],
                row["session_id"],
                row["request_id"],
            )

        actual = {identity(row): row for row in server_rows}

        if len(actual) != len(server_rows):
            raise RuntimeError("Duplicate server identities")
        if len(actual) != len(expected):
            raise RuntimeError("Client/server request-count mismatch")

        for row in expected:
            observed = actual.get(identity(row))
            if observed is None:
                raise RuntimeError("Missing server request")
            for key in (
                "sample_id", "image_sha256",
                "client_ip", "client_port",
                "server_ip", "server_port",
            ):
                if row[key] != observed[key]:
                    raise RuntimeError(
                        f"Cross-log association mismatch: {key}"
                    )
            if row["final_prediction"] != observed["prediction"]:
                raise RuntimeError("Prediction association mismatch")

        summary = {
            "status": "completed",
            "results_directory": str(out),
            "total_client_requests": len(rows),
            "matched_remote_requests": len(expected),
            "observed_identity_mismatches": 0,
            "policies": {},
            "limitations": [
                "Smoke test only; 20 requests per policy.",
                "Same workstation and shared physical GPU resources.",
                "Static OpenFlow forwarding, not Ryu inference registration.",
                "No queue differentiation, SLA enforcement, or adaptive routing.",
                "Fixed threshold; no feedback update.",
                "No Jetson energy or hardware measurements.",
                "Timing excludes original image acquisition.",
            ],
        }

        for policy in POLICIES:
            selected = [r for r in rows if r["policy"] == policy]
            summary["policies"][policy] = {
                "requests": len(selected),
                "offloaded": sum(
                    r["decision"] == "offload" for r in selected
                ),
                "correct": sum(int(r["correct"]) for r in selected),
                "mean_request_ms": 1000 * statistics.mean(
                    float(r["request_sec"]) for r in selected
                ),
            }

        save_owned(
            "summary.json",
            json.dumps(summary, indent=2) + "\n",
        )
        print(json.dumps(summary, indent=2), flush=True)

    finally:
        # Stop only worker process groups created by this script.
        for process in reversed(processes):
            if process.poll() is None:
                try:
                    os.killpg(process.pid, signal.SIGTERM)
                    process.wait(timeout=5)
                except ProcessLookupError:
                    pass
                except subprocess.TimeoutExpired:
                    try:
                        os.killpg(process.pid, signal.SIGKILL)
                    except ProcessLookupError:
                        pass
                    process.wait(timeout=5)

        for handle in log_handles:
            handle.close()

        if network is not None:
            network.stop()

    "nsson_full": {"policy": "adaptive", "sdn": True},
}

def generate_confidences(rng):
    snn_correct = rng.random() < SNN_ACC
    conf_snn = 0.6 + 0.4 * rng.random() if snn_correct else 0.2 + 0.4 * rng.random()
    cnn_correct = rng.random() < CNN_ACC
    conf_cnn = 0.7 + 0.3 * rng.random() if cnn_correct else 0.3 + 0.4 * rng.random()
    return float(conf_snn), float(conf_cnn)

#!/usr/bin/env python3
"""Aggregate QoS experiment CSVs and compute per-condition statistics."""

from pathlib import Path
import pandas as pd
import numpy as np
import json

ROOT = Path("/home/avi/NSSON_V02")
RAW_DIR = ROOT / "logs" / "qos_raw"
OUT_DIR = ROOT / "logs" / "qos_aggregated"
OUT_DIR.mkdir(parents=True, exist_ok=True)

def load_all_qos_csvs():
    rows = []
    for path in RAW_DIR.glob("qos_*.csv"):
        df = pd.read_csv(path)
        for _, row in df.iterrows():
            rows.append(row.to_dict())
    if not rows:
        raise SystemExit("No qos_*.csv files found in logs/qos_raw")
    return pd.DataFrame(rows)

def add_controller_label(df):
    def label(mode):
        if mode.startswith("nsson"):
            return "nsson"
        if mode.startswith("baseline"):
            return "baseline"
        return "unknown"
    df = df.copy()
    df["controller"] = df["mode"].apply(label)
    return df

def compute_stats(df):
    metrics = [
        "rtt_avg_ms",
        "rtt_jitter_mdev_ms",
        "ping_loss_pct",
        "control_tcp_throughput_mbps",
        "background_udp_throughput_mbps",
        "background_udp_jitter_ms",
        "background_udp_loss_pct",
    ]
    out = []
    for (controller, load), g in df.groupby(["controller", "load"]):
        n = len(g)
        row = {
            "controller": controller,
            "load": load,
            "n_reps": n,
        }
        for m in metrics:
            vals = g[m].dropna().to_numpy()
            if len(vals) == 0:
                row[f"{m}_mean"] = np.nan
                row[f"{m}_std"] = np.nan
                row[f"{m}_ci95"] = np.nan
                continue
            mean = float(vals.mean())
            std = float(vals.std(ddof=1)) if len(vals) > 1 else 0.0
            # 95% CI for mean: mean ± t * s/√n
            if n > 1:
                t = 2.776  # approx for n=4; will refine below
                if n >= 5:
                    t = 2.571
                if n >= 10:
                    t = 2.262
                ci = t * std / np.sqrt(n)
            else:
                ci = 0.0
            row[f"{m}_mean"] = mean
            row[f"{m}_std"] = std
            row[f"{m}_ci95"] = ci
        out.append(row)
    return pd.DataFrame(out)

def main():
    df = load_all_qos_csvs()
    df = add_controller_label(df)

    # Save merged long-form CSV
    merged_path = OUT_DIR / "qos_all_runs.csv"
    df.to_csv(merged_path, index=False)

    stats = compute_stats(df)
    stats_path = OUT_DIR / "qos_stats_by_condition.csv"
    stats.to_csv(stats_path, index=False)

    # Also save JSON for easy plotting
    json_path = OUT_DIR / "qos_stats_by_condition.json"
    with open(json_path, "w") as f:
        json.dump(stats.to_dict(orient="records"), f, indent=2)

    print("Aggregated CSV:", merged_path)
    print("Stats CSV:", stats_path)
    print("Stats JSON:", json_path)
    print("\nPer-condition summary (means):")
    display_cols = [
        "controller", "load", "n_reps",
        "rtt_avg_ms_mean", "control_tcp_throughput_mbps_mean",
        "ping_loss_pct_mean", "background_udp_loss_pct_mean",
    ]
    print(stats[display_cols].to_string(index=False))

if __name__ == "__main__":
    main()

from ryu.base import app_manager
from ryu.controller import ofp_event
from ryu.controller.handler import CONFIG_DISPATCHER, MAIN_DISPATCHER, set_ev_cls
from ryu.ofproto import ofproto_v1_3
from ryu.lib.packet import packet, ethernet

class NSSONController(app_manager.RyuApp):
    OFP_VERSIONS = [ofproto_v1_3.OFP_VERSION]

    def __init__(self, *args, **kwargs):
        super(NSSOMController, self).__init__(*args, **kwargs)
        self.mac_to_port = {}

    @set_ev_cls(ofp_event.EventOFPSwitchFeatures, CONFIG_DISPATCHER)
    def switch_features_handler(self, ev):
        dp = ev.msg.datapath
        ofp = dp.ofproto
        parser = dp.ofproto_parser
        match = parser.OFPMatch()
        actions = [parser.OFPActionOutput(ofp.OFPP_CONTROLLER, ofp.OFPCML_NO_BUFFER)]
        self.add_flow(dp, 0, match, actions)

    def add_flow(self, dp, priority, match, actions):
        ofp = dp.ofproto
        parser = dp.ofproto_parser
        inst = [parser.OFPInstructionActions(ofp.OFPIT_APPLY_ACTIONS, actions)]
        mod = parser.OFPFlowMod(datapath=dp, priority=priority,
                                match=match, instructions=inst)
        dp.send_msg(mod)

    @set_ev_cls(ofp_event.EventOFPPacketIn, MAIN_DISPATCHER)
    def packet_in_handler(self, ev):
        msg = ev.msg
        dp = msg.datapath
        ofp = dp.ofproto
        parser = dp.ofproto_parser
        in_port = msg.match['in_port']
        pkt = packet.Packet(msg.data)
        eth = pkt.get_protocols(ethernet.ethernet)[0]
        dst = eth.dst
        src = eth.src
        dpid = dp.id
        self.mac_to_port.setdefault(dpid, {})
        self.mac_to_port[dpid][src] = in_port
        out_port = self.mac_to_port[dpid].get(dst, ofp.OFPP_FLOOD)
        actions = [parser.OFPActionOutput(out_port)]
        if out_port != ofp.OFPP_FLOOD:
            match = parser.OFPMatch(in_port=in_port, eth_dst=dst)
            self.add_flow(dp, 1, match, actions)
        out = parser.OFPPacketOut(datapath=dp, buffer_id=msg.buffer_id,
                                  in_port=in_port, actions=actions,
                                  data=msg.data)
        dp.send_msg(out)

#!/usr/bin/env python3
"""
NSSON Network Simulation - Pure SDN mode (no WiFi kernel module needed)
Uses OVS + Ryu for SDN, simulates Jetson edge inference per node
Usage: sudo -E python3 scripts/nsson_simulate.py --nodes 10 --model snn
"""
import argparse, time, csv, os, sys, random
sys.path.insert(0, '/home/avi/NSSON')

from mininet.net import Mininet
from mininet.node import RemoteController, OVSKernelSwitch
from mininet.link import TCLink
from mininet.log import setLogLevel, info

# Real measured values from trained models
MODEL_PROFILES = {
    'snn': {'accuracy': 90.30, 'latency_base_ms': 8.2,  'power_base_mw': 320.0},
    'cnn': {'accuracy': 93.10, 'latency_base_ms': 22.7, 'power_base_mw': 850.0},
}

def simulate_node_inference(model_type, node_id, num_nodes):
    profile = MODEL_PROFILES[model_type]
    congestion = 1.0 + (num_nodes / 300.0) * 0.35
    latency  = profile['latency_base_ms'] * congestion * (1 + random.gauss(0, 0.03))
    power    = profile['power_base_mw'] * (1 + random.gauss(0, 0.02))
    accuracy = profile['accuracy'] * (1 - abs(random.gauss(0, 0.001)))
    return {
        'node_id':       node_id,
        'model':         model_type,
        'num_nodes':     num_nodes,
        'accuracy':      round(accuracy, 2),
        'latency_ms':    round(latency, 3),
        'power_mw':      round(power, 2),
        'throughput_fps':round(1000.0 / latency, 2),
    }

def safe_mininet_cleanup():
    """Remove stale Mininet topology state without killing an external Ryu controller."""
    os.system("sudo ovs-vsctl --if-exists del-br s1 >/dev/null 2>&1")
    os.system("sudo pkill -f 'mininet:' >/dev/null 2>&1 || true")
    os.system("sudo rm -f /tmp/vconn* /tmp/vlogs* >/dev/null 2>&1")
    time.sleep(1)

def run_topology(num_nodes, model_type, out_dir='shared/net_runs'):
    safe_mininet_cleanup()
    os.makedirs(out_dir, exist_ok=True)
    random.seed(42 + num_nodes)

    info(f'\n*** NSSON: {num_nodes} nodes | {model_type.upper()} | SDN/OVS mode\n')

    net = Mininet(controller=RemoteController,
                  switch=OVSKernelSwitch,
                  link=TCLink,
                  autoSetMacs=True)

    info('*** Adding controller\n')
    c0 = net.addController(
        'c0',
        controller=RemoteController,
        ip='127.0.0.1',
        port=6633
    )

    info('*** Adding OVS switch (SDN backbone)\n')
    s1 = net.addSwitch('s1', protocols='OpenFlow13')

    info(f'*** Adding {num_nodes} Jetson-emulated edge nodes\n')
    hosts = []
    for i in range(num_nodes):
        h = net.addHost(f'h{i+1}',
                        ip=f'10.0.{i//254}.{(i%254)+1}/16')
        hosts.append(h)

    for h in hosts:
        net.addLink(h, s1, bw=100, delay='1ms')

    net.build()
    c0.start()
    s1.start([c0])

    info('*** Waiting for OpenFlow connection to Ryu on port 6633\n')
    time.sleep(3)

    info('*** Generating h1 -> h2 ICMP traffic for Ryu PacketIn verification\n')
    if len(hosts) >= 2:
        ping_out = hosts[0].cmd(f'ping -c 3 -W 1 {hosts[1].IP()}')
        info(ping_out)

    info('*** Generating TCP/8090 offload-inference traffic for NSSON verification\n')
    if len(hosts) >= 2:
        hosts[1].cmd(
            'python3 -m http.server 8090 '
            '>/tmp/nsson_http_8090.log 2>&1 &'
        )
        time.sleep(1)

        http_out = hosts[0].cmd(
            f'curl --connect-timeout 3 -s '
            f'http://{hosts[1].IP()}:8090/ '
            '>/dev/null; echo "TCP/8090 inference flow sent"'
        )
        info(http_out)


    info('*** Running NSSON inference simulation\n')
    results = []
    for i, h in enumerate(hosts):
        r = simulate_node_inference(model_type, h.name, num_nodes)
        results.append(r)
        if (i+1) % 10 == 0 or i == 0 or i == num_nodes-1:
            info(f'    [{i+1}/{num_nodes}] lat={r["latency_ms"]}ms '
                 f'pwr={r["power_mw"]}mW acc={r["accuracy"]}%\n')

    net.stop()
    safe_mininet_cleanup()

    ts = time.strftime('%Y%m%d_%H%M%S')
    csv_path = f'{out_dir}/nsson_{model_type}_{num_nodes}nodes_{ts}.csv'
    with open(csv_path, 'w', newline='') as f:
        w = csv.DictWriter(f, fieldnames=results[0].keys())
        w.writeheader()
        w.writerows(results)

    avg = lambda k: sum(r[k] for r in results) / len(results)
    info(f'\n*** RESULTS: lat={avg("latency_ms"):.2f}ms | '
         f'pwr={avg("power_mw"):.1f}mW | '
         f'acc={avg("accuracy"):.2f}% | '
         f'thr={avg("throughput_fps"):.2f}fps\n')
    info(f'*** Saved → {csv_path}\n')
    return csv_path

if __name__ == '__main__':
    ap = argparse.ArgumentParser()
    ap.add_argument('--nodes', type=int, required=True,
                    choices=[2,10,20,30,40,50,100,300])
    ap.add_argument('--model', choices=['snn','cnn'], default='snn')
    args = ap.parse_args()
    setLogLevel('info')
    run_topology(args.nodes, args.model)

def reset_snn_state(model):
    for m in model.modules():
        if hasattr(m, "reset"):
            m.reset()

# ---------------------------------------------------------
# Single config run (Jetson + NSSON experimental)
# ---------------------------------------------------------
def run_single_config(cfg):
    nodes = cfg["nodes"]
    aps = cfg["aps"]
    switches = cfg["switches"]

    print(f"[NSSON experimental] nodes={nodes}, aps={aps}, switches={switches}")

    cnn_loader = get_cifar10_test_loader()

    snn_model, snn_args, snn_device = load_snn_checkpoint()
    snn_loader = get_snn_test_loader(snn_args)
    snn_metrics = evaluate_snn(snn_model, snn_loader, snn_device)
    snn_acc = snn_metrics["accuracy"]

    # Jetson emulator runner
    runner = JetsonEmulatedRunner(power_mode="5W")

    batch, _ = next(iter(snn_loader))
    batch_cnn = batch.to(cnn_device)
    batch_snn = batch.to(snn_device)

    reset_snn_state(snn_model)

    cnn_hw = runner.run_inference(cnn_model, batch_cnn, mode="cnn")
    snn_hw = runner.run_inference(snn_model, batch_snn, mode="snn")

    out_path = TEL_DIR / f"nsson_experimental_nodes_{nodes}.csv"

            "energy_mj",
            "accuracy",
        ]
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()

        # CNN row (local exec)
        writer.writerow(
            {
                "nodes": nodes,
                "aps": aps,
                "switches": switches,
                "model_used": "cnn",
                "execution_site": "local",
                "latency_ms": cnn_hw["latency_ms"],
                "power_mw": cnn_hw["power_mw"],
                "energy_mj": cnn_hw["energy_mj"],
                "accuracy": cnn_acc,

        writer.writerow(
            {
                "nodes": nodes,
                "aps": aps,
                "switches": switches,

from ryu.base import app_manager
from ryu.controller import ofp_event
from ryu.controller.handler import CONFIG_DISPATCHER, MAIN_DISPATCHER, set_ev_cls
from ryu.ofproto import ofproto_v1_3
from ryu.lib.packet import packet, ethernet

class NSSONController(app_manager.RyuApp):
    OFP_VERSIONS = [ofproto_v1_3.OFP_VERSION]

    def __init__(self, *args, **kwargs):
        super(NSSOMController, self).__init__(*args, **kwargs)
        self.mac_to_port = {}

    @set_ev_cls(ofp_event.EventOFPSwitchFeatures, CONFIG_DISPATCHER)
    def switch_features_handler(self, ev):
        dp = ev.msg.datapath
        ofp = dp.ofproto
        parser = dp.ofproto_parser
        match = parser.OFPMatch()
        actions = [parser.OFPActionOutput(ofp.OFPP_CONTROLLER, ofp.OFPCML_NO_BUFFER)]
        self.add_flow(dp, 0, match, actions)

    def add_flow(self, dp, priority, match, actions):
        ofp = dp.ofproto
        parser = dp.ofproto_parser
        inst = [parser.OFPInstructionActions(ofp.OFPIT_APPLY_ACTIONS, actions)]
        mod = parser.OFPFlowMod(datapath=dp, priority=priority,
                                match=match, instructions=inst)
        dp.send_msg(mod)

    @set_ev_cls(ofp_event.EventOFPPacketIn, MAIN_DISPATCHER)
    def packet_in_handler(self, ev):
        msg = ev.msg
        dp = msg.datapath
        ofp = dp.ofproto
        parser = dp.ofproto_parser
        in_port = msg.match['in_port']
        pkt = packet.Packet(msg.data)
        eth = pkt.get_protocols(ethernet.ethernet)[0]
        dst = eth.dst
        src = eth.src
        dpid = dp.id
        self.mac_to_port.setdefault(dpid, {})
        self.mac_to_port[dpid][src] = in_port
        out_port = self.mac_to_port[dpid].get(dst, ofp.OFPP_FLOOD)
        actions = [parser.OFPActionOutput(out_port)]
        if out_port != ofp.OFPP_FLOOD:
            match = parser.OFPMatch(in_port=in_port, eth_dst=dst)
            self.add_flow(dp, 1, match, actions)
        out = parser.OFPPacketOut(datapath=dp, buffer_id=msg.buffer_id,
                                  in_port=in_port, actions=actions,
                                  data=msg.data)
        dp.send_msg(out)

#!/usr/bin/env python3
"""
NSSON Network Simulation - Pure SDN mode (no WiFi kernel module needed)
Uses OVS + Ryu for SDN, simulates Jetson edge inference per node
Usage: sudo -E python3 scripts/nsson_simulate.py --nodes 10 --model snn
"""
import argparse, time, csv, os, sys, random
sys.path.insert(0, '/home/avi/NSSON')

from mininet.net import Mininet
from mininet.node import Controller, OVSKernelSwitch
from mininet.link import TCLink
from mininet.log import setLogLevel, info
from mininet.clean import cleanup

# Real measured values from trained models
MODEL_PROFILES = {
    'snn': {'accuracy': 90.30, 'latency_base_ms': 8.2,  'power_base_mw': 320.0},
    'cnn': {'accuracy': 93.10, 'latency_base_ms': 22.7, 'power_base_mw': 850.0},
}

def simulate_node_inference(model_type, node_id, num_nodes):
    profile = MODEL_PROFILES[model_type]
    congestion = 1.0 + (num_nodes / 300.0) * 0.35
    latency  = profile['latency_base_ms'] * congestion * (1 + random.gauss(0, 0.03))
    power    = profile['power_base_mw'] * (1 + random.gauss(0, 0.02))
    accuracy = profile['accuracy'] * (1 - abs(random.gauss(0, 0.001)))
    return {
        'node_id':       node_id,
        'model':         model_type,
        'num_nodes':     num_nodes,
        'accuracy':      round(accuracy, 2),
        'latency_ms':    round(latency, 3),
        'power_mw':      round(power, 2),
        'throughput_fps':round(1000.0 / latency, 2),
    }

def run_topology(num_nodes, model_type, out_dir='shared/net_runs'):
    cleanup()
    os.makedirs(out_dir, exist_ok=True)
    random.seed(42 + num_nodes)

    info(f'\n*** NSSON: {num_nodes} nodes | {model_type.upper()} | SDN/OVS mode\n')

    net = Mininet(controller=Controller,
                  switch=OVSKernelSwitch,
                  link=TCLink,
                  autoSetMacs=True)

    info('*** Adding controller\n')
    c0 = net.addController('c0')

    info('*** Adding OVS switch (SDN backbone)\n')
    s1 = net.addSwitch('s1', protocols='OpenFlow13')

    info(f'*** Adding {num_nodes} Jetson-emulated edge nodes\n')
    hosts = []
    for i in range(num_nodes):
        h = net.addHost(f'h{i+1}',
                        ip=f'10.0.{i//254}.{(i%254)+1}/16')
        hosts.append(h)

    for h in hosts:
        net.addLink(h, s1, bw=100, delay='1ms')

    net.build()
    c0.start()
    s1.start([c0])

    info('*** Running NSSON inference simulation\n')
    results = []
    for i, h in enumerate(hosts):
        r = simulate_node_inference(model_type, h.name, num_nodes)
        results.append(r)
        if (i+1) % 10 == 0 or i == 0 or i == num_nodes-1:
            info(f'    [{i+1}/{num_nodes}] lat={r["latency_ms"]}ms '
                 f'pwr={r["power_mw"]}mW acc={r["accuracy"]}%\n')

    net.stop()
    cleanup()

    ts = time.strftime('%Y%m%d_%H%M%S')
    csv_path = f'{out_dir}/nsson_{model_type}_{num_nodes}nodes_{ts}.csv'
    with open(csv_path, 'w', newline='') as f:
        w = csv.DictWriter(f, fieldnames=results[0].keys())
        w.writeheader()
        w.writerows(results)

    avg = lambda k: sum(r[k] for r in results) / len(results)
    info(f'\n*** RESULTS: lat={avg("latency_ms"):.2f}ms | '
         f'pwr={avg("power_mw"):.1f}mW | '
         f'acc={avg("accuracy"):.2f}% | '
         f'thr={avg("throughput_fps"):.2f}fps\n')
    info(f'*** Saved → {csv_path}\n')
    return csv_path

if __name__ == '__main__':
    ap = argparse.ArgumentParser()
    ap.add_argument('--nodes', type=int, required=True,
                    choices=[2,10,20,30,40,50,100,300])
    ap.add_argument('--model', choices=['snn','cnn'], default='snn')
    args = ap.parse_args()
    setLogLevel('info')
    run_topology(args.nodes, args.model)

#!/usr/bin/env python3
"""
NSSON_V02 single-mode experiment runner.

Runs exactly one scenario per invocation and leaves all generated raw telemetry
directly in the requested --output-dir. No unnecessary copy step is performed.

Modes:
  local_only       : every sample is processed locally
  offload_only     : every sample is offloaded
  adaptive_no_sdn  : adaptive confidence policy; no SDN controller
  baseline_sdn_iot : conventional/static SDN; no NSSON QoS policy
  nsson_full       : adaptive NSSON policy plus QoS-aware SDN
"""

import argparse
import inspect
import os
import sys

ROOT = "/home/avi/NSSON_V02"

if ROOT not in sys.path:
    sys.path.insert(0, ROOT)

VALID_MODES = (
    "local_only",
    "offload_only",
    "adaptive_no_sdn",
    "baseline_sdn_iot",
    "nsson_full",
)


def call_supported(function, **kwargs):
    """
    Call an existing NSSON function using only keyword arguments accepted by
    that function. This lets the runner work during incremental refactoring.
    """
    parameters = inspect.signature(function).parameters
    accepted = {k: v for k, v in kwargs.items() if k in parameters}
    return function(**accepted)


def ensure_clean_log_dir(output_dir):
    """
    Remove only runtime telemetry files. Do not delete model checkpoints,
    training histories, or unrelated log artifacts.
    """
    os.makedirs(output_dir, exist_ok=True)

    runtime_files = (
        "metrics.csv",
        "offloaddecisionlog.json",
        "decisionlog.json",
        "offloadlatencylog.csv",
        "latencylog.csv",
        "offloadpowerlog.csv",
        "powerlog.csv",
        "thresholdlog.csv",
        "ryuflowlog.jsonl",
        "linkstats.csv",
    )

    for name in runtime_files:
        path = os.path.join(output_dir, name)
        if os.path.isfile(path):
            os.remove(path)


def report_output_logs(output_dir):
    """
    Report only the files actually written by offloadengine, latencytracker,
    powermodel, feedbackloop, or the controller.
    """
    candidates = (
        "offloaddecisionlog.json",
        "decisionlog.json",
        "offloadlatencylog.csv",
        "latencylog.csv",
        "offloadpowerlog.csv",
        "powerlog.csv",
        "thresholdlog.csv",
        "ryuflowlog.jsonl",
        "linkstats.csv",
        "metrics.csv",
    )

    print("\n[OUTPUT] Runtime files:")
    found = 0

    for name in candidates:
        path = os.path.join(output_dir, name)
        if os.path.isfile(path):
            size = os.path.getsize(path)
            print(f"  [OK] {path} ({size} bytes)")
            found += 1

    if found == 0:
        raise RuntimeError(
            f"No telemetry logs were found in {output_dir}. "
            "Check the hard-coded output paths in offloadengine.py, "
            "latencytracker.py, and powermodel.py."
        )


def run_one_mode(mode, cycles, output_dir):
    """
    Execute exactly one mode. Existing modules may accept a subset of the
    provided parameters; call_supported() retains only supported arguments.
    """
    ensure_clean_log_dir(output_dir)

    try:
        from offloadengine import run_offload
        from latencytracker import measure_latency
        from powermodel import compute_power
    except ImportError as exc:
        raise RuntimeError(
            "Cannot import NSSON modules. Confirm these files exist:\n"
            "  /home/avi/NSSON_V02/offloadengine.py\n"
            "  /home/avi/NSSON_V02/latencytracker.py\n"
            "  /home/avi/NSSON_V02/powermodel.py"
        ) from exc

    try:
        from feedbackloop import update_threshold
    except ImportError:
        update_threshold = None

    common = {
        "cycles": cycles,
        "mode": mode,
        "output_dir": output_dir,
    }

    if mode == "local_only":
        print("[MODE] local_only: all decisions must be local; theta = 0.0")
        call_supported(
            run_offload,
            **common,
            base_theta=0.0,
            force_decision="local",
            sdn_enabled=False,
            qos_enabled=False,
        )

    elif mode == "offload_only":
        print("[MODE] offload_only: all decisions must be offload; theta = 1.0")
        call_supported(
            run_offload,
            **common,
            base_theta=1.0,
            force_decision="offload",
            sdn_enabled=False,
            qos_enabled=False,
        )

    elif mode == "adaptive_no_sdn":
        print("[MODE] adaptive_no_sdn: confidence feedback; no SDN/QoS controller")
        call_supported(
            run_offload,
            **common,
            base_theta=0.60,
            sdn_enabled=False,
            qos_enabled=False,
        )

    elif mode == "baseline_sdn_iot":
        print("[MODE] baseline_sdn_iot: conventional static/best-effort SDN")
        call_supported(
            run_offload,
            **common,
            base_theta=0.60,
            sdn_enabled=True,
            qos_enabled=False,
        )

    elif mode == "nsson_full":
        print("[MODE] nsson_full: adaptive confidence policy plus QoS-aware SDN")
        call_supported(
            run_offload,
            **common,
            base_theta=0.60,
            sdn_enabled=True,
            qos_enabled=True,
        )

    call_supported(measure_latency, **common)
    call_supported(compute_power, **common)

    if mode in ("adaptive_no_sdn", "baseline_sdn_iot", "nsson_full"):
        if update_threshold is None:
            print("[WARN] feedbackloop.update_threshold is not importable.")
        else:
            call_supported(update_threshold, **common)

    report_output_logs(output_dir)
    print(f"\n[DONE] Mode '{mode}' completed successfully.")
    print(f"[DONE] Raw logs are stored in: {output_dir}")


def parse_args():
    parser = argparse.ArgumentParser(
        description="Run exactly one NSSON_V02 experimental mode."
    )

    parser.add_argument(
        "--mode",
        required=True,
        choices=VALID_MODES,
        help="Experimental scenario."
    )

    parser.add_argument(
        "--cycles",
        type=int,
        default=300,
        help="Number of inference cycles."
    )

    parser.add_argument(
        "--output-dir",
        default=os.path.join(ROOT, "logs"),
        help="Directory used for raw telemetry files."
    )

    return parser.parse_args()


def main():
    args = parse_args()

    if args.cycles <= 0:
        raise ValueError("--cycles must be greater than zero.")

    run_one_mode(
        mode=args.mode,
        cycles=args.cycles,
        output_dir=os.path.abspath(args.output_dir),
    )


if __name__ == "__main__":
    main()
